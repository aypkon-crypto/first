////////////////////////////////////////////////// // PHONE INTERFACE FUNCTIONS // ////////////////////////////////////////////////// #pragma region Phone Helpers /** * @brief Reads touch messages from the phone web page over USB serial * * Protocol (one message per line): * - "T," finger on the phone pad, x/y in 0-1000 (y = 0 at the bottom) * - "U" finger lifted * * Also sends a short status line to the phone's Logs panel twice per second. * * Why: Non-blocking, so Mozzi's audio timing is never disturbed */ void readPhone() { static char buf[16]; static uint8_t len = 0; while (Serial.available()) { char c = Serial.read(); if (c == '\n') { buf[len] = '\0'; handlePhoneMessage(buf); len = 0; } else if (len < sizeof(buf) - 1) { buf[len++] = c; } } // Release the note if the phone goes quiet (e.g. cable unplugged) if (phoneTouching && --phoneTimeout == 0) { phoneTouching = false; phoneTimeouts++; } // Status report twice per second (short, so Serial never blocks the audio) static uint16_t reportCount = 0; if (++reportCount >= CONTROL_RATE / 2) { reportCount = 0; Serial.print(F("ok=")); Serial.print(phoneMsgCount); Serial.print(F(" bad=")); Serial.print(phoneBadCount); Serial.print(F(" tmo=")); Serial.print(phoneTimeouts); Serial.print(F(" touch=")); Serial.print(phoneTouching); Serial.print(F(" ph=")); Serial.print(phoneX); Serial.print(','); Serial.print(phoneY); Serial.print(F(" pad=")); Serial.print(xVal); Serial.print(','); Serial.println(yVal); } } /** * @brief Maps one phone message onto the touchpad's calibrated range * @param msg Null-terminated message without the newline */ void handlePhoneMessage(char* msg) { if (msg[0] == 'T') { char* comma = strchr(msg, ','); if (!comma) { phoneBadCount++; return; } int x = constrain(atoi(msg + 1), 0, 1000); int y = constrain(atoi(comma + 1), 0, 1000); phoneX = map(x, 0, 1000, TOUCHPAD_X_MIN, TOUCHPAD_X_MAX); phoneY = map(y, 0, 1000, TOUCHPAD_Y_MIN, TOUCHPAD_Y_MAX); phoneY = min(phoneY, TOUCHPAD_EDGE_DETECT - 1); // Always counts as "finger on" phoneTouching = true; phoneTimeout = PHONE_TIMEOUT_CYCLES; phoneMsgCount++; } else if (msg[0] == 'U') { phoneTouching = false; phoneMsgCount++; } else { phoneBadCount++; } } #pragma endregion Phone Helpers
